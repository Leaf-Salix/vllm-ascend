227:   /// Scope the IO reorder to bodies of `ForKind::Pipeline` loops. Non-pipelined
228:   /// code is visited recursively but its SeqStmts are left as-is.
229:   ///
230:   /// On exit from a pipeline scope, demote `kind_` to `Sequential` and strip
231:   /// the `pipeline_stages` attr together — the marker has served its purpose
232:   /// (gated this reorder) and must not survive past this pass. The bidirectional
233:   /// invariant `kind == Pipeline ⇔ pipeline_stages attr present` (PipelineLoopValid)
234:   /// guarantees the attr is present on entry, so we strip unconditionally.
235:   /// The PipelineResolved verifier checks the post-condition: no
236:   /// ForKind::Pipeline loops downstream of CanonicalizeIOOrder.
237:   StmtPtr VisitStmt_(const ForStmtPtr& op) override {
238:     const bool is_pipeline = (op->kind_ == ForKind::Pipeline);
239:     const bool saved_overlap = overlap_stores_;
240:     const bool saved_dbc = double_buffer_c_;
241:     if (is_pipeline) {
242:       inside_pipeline_depth_++;
243:       // Default true: stores float to the Store tier (output ping-pong). A loop
244:       // opts into the one-accumulator schedule with overlap_stores=false.
245:       overlap_stores_ = op->GetAttr<bool>(kPipelineOverlapStoresAttr, true);
246:       // Default false: only dbC=2 loops lift stores above all compute (co-live
247:       // two-accumulator schedule). See kPipelineDoubleBufferCAttr.
248:       double_buffer_c_ = op->GetAttr<bool>(kPipelineDoubleBufferCAttr, false);
249:     }
250:     auto visited = IRMutator::VisitStmt_(op);
251:     if (is_pipeline) {
252:       inside_pipeline_depth_--;
253:       overlap_stores_ = saved_overlap;
254:       double_buffer_c_ = saved_dbc;
255:     }
256: 
257:     if (!is_pipeline) return visited;
258:     auto visited_for = std::dynamic_pointer_cast<const ForStmt>(visited);
259:     if (!visited_for) return visited;
260:     auto demoted = MutableCopy(visited_for);
261:     demoted->kind_ = ForKind::Sequential;
262:     // Strip both pipeline markers — they have served their purpose (gated this
263:     // reorder) and must not survive past this pass.
264:     demoted->attrs_ =
265:         StripAttr(StripAttr(StripAttr(demoted->attrs_, kPipelineStagesAttr), kPipelineOverlapStoresAttr),
266:                   kPipelineDoubleBufferCAttr);
267:     return demoted;
268:   }
269: 
270:   StmtPtr VisitStmt_(const SeqStmtsPtr& op) override {
271:     // Recurse first so any nested SeqStmts are reordered bottom-up.
272:     auto visited = IRMutator::VisitStmt_(op);
273:     if (inside_pipeline_depth_ == 0) {
274:       return visited;  // outside any pipeline scope — do not reorder
275:     }
276:     auto seq = std::dynamic_pointer_cast<const SeqStmts>(visited);
277:     if (!seq || seq->stmts_.size() < 2) {
278:       return visited;  // single stmt — nothing to reorder
279:     }
280:     return ReorderRegion(seq);
